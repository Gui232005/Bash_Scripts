#!/bin/bash

cd database
directory="/home/guilherme/Desktop/Bash_Scripts/database"
dt=$(date +%F)

if [ ! -d "$directory" ]; then
    echo "-1"
    exit 1
fi

mkdir -p backups
cd backups
mkdir -p backup_$dt
cd ..
for FILE in *.txt; 
    do echo $FILE
    cp $FILE /home/guilherme/Desktop/Bash_Scripts/database/backups/backup_$dt;
done
